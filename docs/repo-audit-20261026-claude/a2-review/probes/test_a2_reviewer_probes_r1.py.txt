These probes reuse the permanent regression's SYNTHETIC context builders as

ENTRY_FAMILIES = ("TREND", "BREAKOUT", "PULLBACK", "REVERSION", "GRID", "MARTINGALE")


def adverse(world, plan):
    """Deliver ``plan`` entries: {"step", "kw"?, "rebuild"?} against real strategies.

    Returns one row per delivery with the committed action, confirmation_count
    and setup identity; ``rebuild`` reconstructs the strategy object first
    (saved observation state + fresh object), never fabricating an early exit.
    """
    strategy, steps = world
    state = None
    rows = []
    for entry in plan:
        if entry.get("rebuild"):
            strategy = type(strategy)(strategy.spec)
        context = steps[entry["step"]](**(entry.get("kw") or {}))
        proposal = strategy.evaluate(context, state)
        state = commit_observation(state, proposal)
        rows.append({
            "action": proposal.action,
            "count": state.confirmation_count,
            "setup_id": state.setup.setup_id if state.setup is not None else None,
            "expiry": state.setup.expires_at_nano if state.setup is not None else None,
        })
    return rows


@pytest.mark.parametrize("mode", ("MINUTE", "DAILY"))
def test_c1_expired_count2_restart_never_lends_credit_and_rebuild(mode):
    # Pre-fix discriminator: stale 2 -> 3 at the rebuilt arming OPENed on the
    # new period's first recross ("ra", row index 5). Fixed: WAIT through ra,
    # OPEN only on the new period's third distinct qualifying bar ("rb").
    rows = adverse(build_world("REVERSION", mode), [
        {"step": "s0"}, {"step": "s1"},
        {"step": "mid"},                      # distinct post-expiry middle bar: 0
        {"step": "arm", "rebuild": True},     # rebuilt object, fresh arming: 1
        {"step": "arm"},                      # same-bar redelivery holds 1
        {"step": "ra"},                       # first recross of NEW period: 2, no OPEN
        {"step": "ra"},
        {"step": "rb"},
    ])
    assert [row["count"] for row in rows] == [1, 2, 0, 1, 1, 2, 2, 3], rows
    assert [row["action"] for row in rows] == ["WAIT"] * 7 + ["OPEN"], rows
    assert rows[1]["setup_id"] != rows[3]["setup_id"]  # genuinely new period


@pytest.mark.parametrize("family", ENTRY_FAMILIES)
@pytest.mark.parametrize("mode", ("MINUTE",))
def test_c2_expired_same_bar_redelivery_never_opens_nor_keeps_credit(family, mode):
    rows = adverse(build_world(family, mode), [{"step": "s0"}, {"step": "s1"}])
    assert rows[0]["count"] == 1 and rows[1]["count"] == 2, rows
    expiry = rows[1]["expiry"]
    assert expiry is not None
    late = adverse(build_world(family, mode), [
        {"step": "s0"}, {"step": "s1"},
        {"step": "s1", "kw": {"now_override": expiry + 1}},
    ])[2]
    # Pre-fix every family reached 3 here (several OPENed); fixed: never OPEN,
    # never more than a fresh period's first observation, and a count of 1 must
    # belong to a NEW observation period, not the expired one.
    assert late["action"] != "OPEN", late
    assert late["count"] <= 1, late
    if late["count"] == 1:
        assert late["setup_id"] is not None
        assert late["setup_id"] != rows[1]["setup_id"], late

@pytest.mark.parametrize("mode", ("MINUTE", "DAILY"))
@pytest.mark.parametrize("family", ("REVERSION", "TREND"))
def test_c3_gate_failure_zeroes_first_same_bar_recovery_does_not_recount(family, mode):
    # Admission failure resets to 0; the same bar later qualifying again must
    # not re-count (REVERSION pre-fix re-counted it and OPENed one bar early:
    # 1,0,1,2,3-OPEN). Fixed: 1,0,0,1,2 with no OPEN anywhere.
    rows = adverse(build_world(family, mode), [
        {"step": "s0"},
        {"step": "s1", "kw": {"gate_ok": False}},
        {"step": "s1"},
        {"step": "s2"},
        {"step": "s3"},
    ])
    assert [row["count"] for row in rows] == [1, 0, 0, 1, 2], rows
    assert all(row["action"] == "WAIT" for row in rows), rows


def test_c3_breakout_gate_failure_beats_created_restart():
    # A genuine BREAKOUT creation on a gate-failed decision still yields 0
    # (gate-failure priority over created->1); the same bar recovering holds 0;
    # the OPEN then needs three further distinct qualifying bars.
    rows = adverse(build_world("BREAKOUT", "MINUTE"), [
        {"step": "s0", "kw": {"gate_ok": False}},
        {"step": "s0"},
        {"step": "s1"},
        {"step": "s2"},
        {"step": "s3"},
    ])
    assert [row["count"] for row in rows] == [0, 0, 1, 2, 3], rows
    assert [row["action"] for row in rows] == ["WAIT"] * 4 + ["OPEN"], rows
